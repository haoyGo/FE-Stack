# 大文件上传原理

## 核心原理

大文件上传通过**文件分片**解决超时和断点续传问题。

### 基本流程

```
1. 文件切片：将大文件按固定大小（如 5MB）切分成多个小块
2. 计算指纹：用 Hash 算法生成文件唯一标识（用于秒传和去重）
3. 并发上传：同时上传多个分片，提高效率
4. 断点续传：记录已上传分片，失败后只传剩余部分
5. 合并文件：服务端接收所有分片后，按顺序合并成完整文件
```

### 关键技术

- **分片（Chunking）**：`file.slice(start, end)` 切分文件
- **哈希（Hash）**：生成文件指纹，实现秒传
- **并发控制**：限制同时上传数量，避免占用过多资源
- **断点续传**：记录上传状态，中断后继续传输

## 代码实现

### 前端核心代码

```javascript
class FileUploader {
  constructor(options) {
    this.chunkSize = options.chunkSize || 5 * 1024 * 1024; // 5MB
    this.uploadedChunks = new Set(); // 已上传分片
    this.maxConcurrent = options.maxConcurrent || 3; // 并发数
  }

  async uploadFile(file) {
    // 1. 计算文件 Hash（用于秒传和去重）
    const fileHash = await this.calculateHash(file);

    // 2. 检查文件是否已存在（秒传）
    const existed = await this.checkFileExist(fileHash);
    if (existed) {
      console.log("文件已存在，秒传成功");
      return;
    }

    // 3. 切分文件
    const chunks = this.createChunks(file);

    // 4. 获取已上传的分片（断点续传）
    const uploadedList = await this.getUploadedList(fileHash);
    this.uploadedChunks = new Set(uploadedList);

    // 5. 上传分片（并发控制）
    await this.uploadChunks(chunks, fileHash);

    // 6. 通知服务器合并分片
    await this.mergeChunks(fileHash, file.name);
  }

  // 切分文件
  createChunks(file) {
    const chunks = [];
    let start = 0;

    while (start < file.size) {
      const end = Math.min(start + this.chunkSize, file.size);
      chunks.push({
        chunk: file.slice(start, end), // 使用 Blob.slice
        index: chunks.length,
      });
      start = end;
    }

    return chunks;
  }

  // 计算文件 Hash
  async calculateHash(file) {
    // 简化版：实际应使用 spark-md5 或 WebWorker
    return new Promise((resolve) => {
      const reader = new FileReader();
      reader.onload = (e) => {
        const hash = this.simpleHash(e.target.result);
        resolve(hash);
      };
      reader.readAsArrayBuffer(file);
    });
  }

  simpleHash(buffer) {
    let hash = 0;
    const view = new DataView(buffer);
    for (let i = 0; i < view.byteLength; i += 4) {
      hash = (hash << 5) - hash + view.getInt32(i, true);
    }
    return hash.toString(16);
  }

  // 并发上传分片
  async uploadChunks(chunks, fileHash) {
    // 过滤已上传的分片
    const uploadQueue = chunks.filter(
      (item) => !this.uploadedChunks.has(`${fileHash}-${item.index}`)
    );

    const uploadChunk = async (chunkData) => {
      const formData = new FormData();
      formData.append("chunk", chunkData.chunk);
      formData.append("fileHash", fileHash);
      formData.append("chunkIndex", chunkData.index);

      const response = await fetch("/api/upload/chunk", {
        method: "POST",
        body: formData,
      });

      if (response.ok) {
        this.uploadedChunks.add(`${fileHash}-${chunkData.index}`);
      }
    };

    // 并发控制
    const executing = [];
    for (const chunkData of uploadQueue) {
      const promise = uploadChunk(chunkData).then(() => {
        executing.splice(executing.indexOf(promise), 1);
      });

      executing.push(promise);

      // 控制并发数
      if (executing.length >= this.maxConcurrent) {
        await Promise.race(executing);
      }
    }

    await Promise.all(executing);
  }

  async checkFileExist(fileHash) {
    const response = await fetch(`/api/upload/verify`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ fileHash }),
    });
    const data = await response.json();
    return data.exist;
  }

  async getUploadedList(fileHash) {
    const response = await fetch(`/api/upload/uploaded?fileHash=${fileHash}`);
    const data = await response.json();
    return data.uploadedList || [];
  }

  async mergeChunks(fileHash, fileName) {
    await fetch("/api/upload/merge", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ fileHash, fileName }),
    });
  }
}

// 使用
const uploader = new FileUploader({ chunkSize: 5 * 1024 * 1024 });
document.getElementById("fileInput").addEventListener("change", (e) => {
  uploader.uploadFile(e.target.files[0]);
});
```

### 后端核心代码（Node.js）

```javascript
const express = require("express");
const multer = require("multer");
const fs = require("fs-extra");
const path = require("path");

const app = express();
const upload = multer({ dest: "temp/" });

app.use(express.json());

const UPLOAD_DIR = "./uploads";
const TEMP_DIR = "./temp";

// 1. 检查文件是否存在（秒传）
app.post("/api/upload/verify", async (req, res) => {
  const { fileHash } = req.body;
  const filePath = path.join(UPLOAD_DIR, fileHash);

  if (await fs.pathExists(filePath)) {
    return res.json({ exist: true });
  }

  res.json({ exist: false });
});

// 2. 上传分片
app.post("/api/upload/chunk", upload.single("chunk"), async (req, res) => {
  const { fileHash, chunkIndex } = req.body;
  const chunkDir = path.join(TEMP_DIR, fileHash);

  await fs.ensureDir(chunkDir);

  const chunkPath = path.join(chunkDir, `${fileHash}-${chunkIndex}`);
  await fs.move(req.file.path, chunkPath);

  res.json({ success: true });
});

// 3. 查询已上传分片
app.get("/api/upload/uploaded", async (req, res) => {
  const { fileHash } = req.query;
  const chunkDir = path.join(TEMP_DIR, fileHash);

  if (await fs.pathExists(chunkDir)) {
    const uploadedList = await fs.readdir(chunkDir);
    res.json({ uploadedList });
  } else {
    res.json({ uploadedList: [] });
  }
});

// 4. 合并分片
app.post("/api/upload/merge", async (req, res) => {
  const { fileHash, fileName } = req.body;
  const chunkDir = path.join(TEMP_DIR, fileHash);
  const filePath = path.join(UPLOAD_DIR, fileHash);

  // 读取所有分片
  const chunkPaths = await fs.readdir(chunkDir);

  // 按顺序合并
  chunkPaths.sort((a, b) => {
    const indexA = parseInt(a.split("-").pop());
    const indexB = parseInt(b.split("-").pop());
    return indexA - indexB;
  });

  // 创建写入流
  await fs.ensureDir(UPLOAD_DIR);
  const writeStream = fs.createWriteStream(filePath);

  // 依次写入分片
  for (const chunkPath of chunkPaths) {
    const data = await fs.readFile(path.join(chunkDir, chunkPath));
    writeStream.write(data);
  }

  writeStream.end();

  // 清理临时文件
  await fs.remove(chunkDir);

  res.json({ success: true, url: `/files/${fileHash}` });
});

app.listen(3000);
```
